macro round value
    set value + - value 16777216 16777216
endmacro

; round to the nearest whole pixel (1.5 x 2^23 keeps float32 precision at 1.0,
; the 16777216 version above snaps to even numbers)
macro roundpx value
    set value + - value 12582912 12582912
endmacro

macro floor value
    set value + - - value 0.499999 16777216 16777216
endmacro

macro ceil value
    set value value>0 + . 0.99999
    floor value
endmacro

macro calcFontSizes scale
    ; Windows font
    set main_font           os_type==0 scale==1 [2] scale==1.5 [5] [8]
    set list_font           os_type==0 scale==1 [2] scale==1.5 [5] [9]
    set trans_font          os_type==0 scale==1 [2] scale==1.5 [6] [10]
    set header_font         os_type==0 scale==1 [3] scale==1.5 [8] [11]

    set main_font_voffs     os_type==0 scale==1 0.5 scale==1.5 -0.5 -0.5
    set trans_font_voffs    os_type==0 scale==1 0   scale==1.5  0    0
    set header_font_voffs   os_type==0 scale==1 1   scale==1.5  0    0

    ; MacOS font
    set main_font           os_type==1 scale==1 [4] scale==1.5 [7]  [12]
    set list_font           os_type==1 scale==1 [3] scale==1.5 [9]  [13]
    set trans_font          os_type==1 scale==1 [4] scale==1.5 [9]  [15]
    set header_font         os_type==1 scale==1 [6] scale==1.5 [11] [16]

    set main_font_voffs     os_type==1 scale==1 0.5 scale==1.5 -0.5 -0.5
    set trans_font_voffs    os_type==1 scale==1 0   scale==1.5  0    0
    set header_font_voffs   os_type==1 scale==1 1   scale==1.5  2    0

    set main_font           os_type==3 scale==1 [1] scale==1.5 [5] [8]
    set list_font           os_type==3 scale==1 [1] scale==1.5 [4] [8]
    set trans_font          os_type==3 scale==1 [2] scale==1.5 [6] [9]
    set header_font         os_type==3 scale==1 [3] scale==1.5 [9] [12]

    set main_font_voffs     os_type==3 scale==1 0.5 scale==1.5 -0.5 -0.5
    set trans_font_voffs    os_type==3 scale==1 0   scale==1.5  1    1
    set header_font_voffs   os_type==3 scale==1 1   scale==1.5  2    0

    ; Linux font
    set main_font           os_type==2 scale==1 [3] scale==1.5 [8]  [12]
    set list_font           os_type==2 scale==1 [3] scale==1.5 [9]  [13]
    set trans_font          os_type==2 scale==1 [4] scale==1.5 [10] [15]
    set header_font         os_type==2 scale==1 [5] scale==1.5 [11] [16]

    set main_font_voffs     os_type==2 scale==1 0.5 scale==1.5 -0.5 -0.5

macro shrinkLabelAndVolume min_label_len min_vol_len max_shrink_total

	set max_shrink_label - tcp_LabelSize + indent_offs - min_label_len + recmon_w tcp_stereo_pan_align==2 ?width_visible 23 0 0
	set max_shrink_label max_shrink_label<0 0

	set max_shrink_vol - tcp_Volsize min_vol_len
	set max_shrink_vol max_shrink_vol<0 0

    ; Calculate the ratio of max shrink values
    set shrink_ratio / max_shrink_label max_shrink_vol
    
    ; Calculate initial proportional shrink values
    set shrink_vol / max_shrink_total + shrink_ratio 1
    set shrink_label * shrink_vol shrink_ratio
    
    ; Check if either volume or label exceeds its max shrink
	set shrink_label shrink_label>max_shrink_label max_shrink_label
	set shrink_vol shrink_label>max_shrink_label - max_shrink_total max_shrink_label

	set shrink_label shrink_vol>max_shrink_vol - max_shrink_total max_shrink_vol
	set shrink_vol shrink_vol>max_shrink_vol max_shrink_vol

    set did_shrink 0
	set did_shrink max_shrink_total>0 shrink_label<=max_shrink_label shrink_vol<=max_shrink_vol 1
	set did_shrink !tcp_allow_shrink 0

	set tcp_LabelSize 	?did_shrink - . shrink_label
	set tcp_VolSize  	?did_shrink - . shrink_vol
endmacro

; REAPERTIPS MOD: pan, then volume, go to the second row (pan_late / vol_late) when their "Keep"

macro firstRowLate

; Smart folder indent: each extra folder level adds less space than the one before.
; Off = classic linear indent (depth x indent). On = 1, 1.75, 2.31, 2.81 ... units,
; each extra level adds 25% less, never less than half a level.
macro indentUnits out depth
	set out											!smart_folder_indent{0} depth{0} \
													depth{0}<=0 0 \
													depth{0}==1 1 \
													depth{0}==2 1.75 \
													depth{0}==3 2.31 \
													depth{0}==4 2.81 \
													depth{0}==5 3.31 \
													depth{0}==6 3.81 \
													+ 3.81 * 0.5 - depth{0} 6
endmacro

; Accumulated indent in unscaled px, rounded to a whole pixel.
macro indentAcc out indentpx unitsv
	set out											* indentpx{0} unitsv{0}
	roundpx out
endmacro

macro resizeInput is_next_row
    set min_input_w         * scale element{y}<element_h 104 49
    set fill_end            element{y}<element_h main_eff{0} ?labelFlip main_eff{0} first_row_r
    set remainder           tcp_InSize<=49 0 - this_end{0} fill_end
    set remainder           remainder<0 0 ; growing is done by fillRows, after the rows are placed
    set remainder           remainder>0 !tcp_allow_shrink 0
    set adj_width           - s_width remainder
    set min_input_w         element{y}>=element_h !labelFlip !tcp_control_align==1 + . * scale - indent_acc_max indent_acc
    set element		        element{w}>0 ?fill adj_width>min_input_w - . remainder@w
    set this_end			element{w}>0 ?fill adj_width>min_input_w - . remainder
    ; push to the next row only when the input really has to shrink (remainder > 0) and can't keep its
    ; minimum width; at its natural size it stays (the smallest input size went to a third row)
    set this_end			element{w}>0 ?fill remainder{0}>0 adj_width<=min_input_w ?tcp_allow_shrink ?tcp_expand_input !is_next_row  + main_eff{0} 1
endmacro

macro then element width use_padding fill exclude swap
	; apply scale
	set s_width				* width scale

	set prev_end 			!swap + [previous_element] [previous_element{2}]								; find the end of the previous element
	set element 			!swap [prev_end previous_element s_width element_h]								; ...and place the element there

    set is_hidden           ?hide_##element{0} 1 ?exclude 1 0   ; FTC MOD
    set element				!swap ?is_hidden [. . 0 element_h]										        ; apply hide params

	; x-axis
	set element 			!swap ?use_padding !is_hidden + [tcp_padding] 							        ; if x-padding is required, add it
	set this_end			!swap - + + [element] [element{2}] [tcp_padding] [main_sec]						; okay, and what does that make the x-end of this element, realtive to main_sec?
    
    resizeInput 0 ;FTC MOD: Shrink/expand input ***
    
	set element				!swap ?mainCollapse{0} [. . 0 0]

	; y-axis
	set next_row			!swap + [0 element] [0 previous_element{3}] 									; find the y of the next row
	set element 			!swap this_end{0}>main_eff{0} + [main_sec next_row element element] tcp_padding	; move it to the next row
	set this_end			!swap - + + [element] [element{2}] [tcp_padding] [main_sec]						; recalculate this_end, now that the element is on a new row
	
    resizeInput 1 ;FTC MOD: Shrink/expand input ***
    
	set element				!swap this_end{0}>main_eff{0} [. . 0 0]											; ... and cull if there's still not enough main_sec width to fit
	set this_y_end			!swap + + + element{1} element{3} tcp_padding{1} scale==1.5 -0.5 0				; find its bottom
	set element				!swap this_y_end{0}>main_sec{3} [0] !previous_element{2} !previous_element{3} [0] . .	; cull if there's not enough height to fit, or if previous_element was culled
    
    ;FTC MOD Gather info on where rows end
    set first_row_r 		!swap element{y}<element_h this_end{0}
    set any_row_r   		!swap element{w}>0 this_end{0}>any_row_r{0} this_end{0}<=main_eff{0} this_end{0}

	; values readout control
	set valuesRoom			!swap - - main_sec{3} this_y_end{0} tcp_padding{1}								; is the remaining height after this element enough for the values to fit?
	set hideValues			!swap valuesRoom{0}<8 element{3}!=0 1 . .										; if not, and this element isn't already hidden, cull the values

	set previous_element    !swap element
	set main_sec_last 		!swap element{w}>0 this_y_end{0} .
	
endmacro

; REAPERTIPS MOD: "Expand elements to fill row". After the rows are placed, the track name, then a
; volume fader (not a knob), then the record input take the room left at the end of their row.
; The elements after them in that row move right by the same amount.
macro fillEndOf el
    set fill_tmp                el{w}>0 el{y}==fill_y{0} + el{x} el{w} 0 0
    set fill_row_end            fill_tmp{0}>fill_row_end{0} fill_tmp .
endmacro

macro fillShift el
    set el                      el{w}>0 el{y}==fill_y{0} el{x}>fill_x{0} + el [fill_grow{0}] . . .
endmacro

    fillEndOf tcp.infoblock
    fillEndOf tcp.io

    fillEndOf tcp.env

macro calcTcpFlow
    set any_row_r 0
    set first_row_r 0
    set max_info_w / - main_sec{w} * 2 tcp_padding{0} scale
    set tcp_info_Size   tcp_infoblock_width>max_info_w max_info_w tcp_infoblock_width
    set show_info ?tcp_infoblock ?tcp_infoblock_wrap 1 0

    ;    element        size  	        padding?    fill    exclude             swap
    then tcp.recarm     20 			    0           0       0                   0
    then tcp.recmon     15 			    0           0       hide_folder_recmon  0
    then tcp.label 	    tcp_LabelSize 	0           0       0                   0    
    ; volume, pan and IO go here (first row) or after it (second row), see firstRowLate and calcTcpSecs
    then tcp.volume     tcp_VolSize 	0           0       0                   skip_vol_early
    then pan_group	    tcp_PanSize		1           0       0                   skip_pan_early
    ; a kept infoblock or routing button (FX with swap) that doesn't fit in the first row goes back to
    ; its normal place, after the volume
    set info_row1_end   - + + + previous_element{x} previous_element{w} * 2 tcp_padding{0} * tcp_info_Size scale main_sec{x}
    set info_early_now  ?info_early info_row1_end{0}<=main_eff{0} previous_element{y}<element_h 1 0 0 0
    set skip_info_early !info_early_now 1 0
    set skip_info_late  ?info_early_now 1 0
    then tcp.infoblock  tcp_info_Size   1           0       !show_info          skip_info_early
    set io_row1_end     - + + + previous_element{x} previous_element{w} * 2 tcp_padding{0} * keep_btn_w scale main_sec{x}
    set io_early_now    ?io_early io_row1_end{0}<=main_eff{0} previous_element{y}<element_h 1 0 0 0
    set skip_io_early   ?tcp_swap_fx_io 1 !io_early_now 1 0
    set skip_fx_early   !tcp_swap_fx_io 1 !io_early_now 1 0
    set skip_io_a       ?io_early_now 1 tcp_swap_fx_io
    set skip_fx_b       !tcp_swap_fx_io 1 io_early_now
    set skip_io_b       !tcp_swap_fx_io 1 0
    then tcp.io 	    28 			    1           0       show_side_io_button skip_io_early
    then fx_group 	    36 			    1           0       show_side_fx_button skip_fx_early
    then tcp.volume     tcp_VolSize 	1           0       0                   skip_vol_late
    then pan_group	    tcp_PanSize		1           0       0                   skip_pan_late
    then tcp.infoblock  tcp_info_Size   1           0       !show_info          skip_info_late
    then tcp.io 	    28 			    1           0       show_side_io_button skip_io_a
    then fx_group 	    36 			    1           0       show_side_fx_button tcp_swap_fx_io
    then fx_group 	    36 			    1           0       show_side_fx_button skip_fx_b
    then tcp.io 	    28 			    1           0       show_side_io_button skip_io_b
    then tcp.env 	    20 			    1           0       0                   0
    then tcp.recmode    20 			    1           0       0                   0
    then input_group    tcp_InSize		1           1       0                   0
endmacro

macro shrinkEnvLabelAndVolume min_label_len min_vol_len max_shrink_total

	set max_shrink_label - envcp_LabelSize min_label_len
	set max_shrink_label max_shrink_label<0 0

	set max_shrink_vol - envcp_Volsize min_vol_len
	set max_shrink_vol max_shrink_vol<0 0

    ; Calculate the ratio of max shrink values
    set shrink_ratio / max_shrink_label max_shrink_vol
    
    ; Calculate initial proportional shrink values
    set shrink_vol / max_shrink_total + shrink_ratio 1
    set shrink_label * shrink_vol shrink_ratio
    
    ; Check if either volume or label exceeds its max shrink
	set shrink_label shrink_label>max_shrink_label max_shrink_label
	set shrink_vol shrink_label>max_shrink_label - max_shrink_total max_shrink_label

	set shrink_label shrink_vol>max_shrink_vol - max_shrink_total max_shrink_vol
	set shrink_vol shrink_vol>max_shrink_vol max_shrink_vol

    set did_shrink 0
	set did_shrink max_shrink_total>0 shrink_label<=max_shrink_label shrink_vol<=max_shrink_vol 1
	set did_shrink !tcp_allow_shrink 0

	set envcp_LabelSize 	?did_shrink - . shrink_label
	set envcp_VolSize  	    ?did_shrink - . shrink_vol
endmacro

macro envThen element width use_padding
	; apply scale
	set s_width				* width scale

	set prev_end 			+ [previous_element] [previous_element{2}]									; find the end of the previous element
	set element 			[prev_end previous_element s_width element_h]									; ...and place the element there

	; x-axis
	set element 			?use_padding + [envcp_padding] 												; if x-padding is required, add it
	set this_end			- + + [element] [element{2}] [envcp_padding] [main_sec]						; okay, and what does that make the x-end of this element, realtive to main_sec?

	; y-axis
	set next_row			+ [0 element] [0 previous_element{3}] 										; find the y of the next row
	set element 			this_end{0}>main_sec{2} + [main_sec next_row element element] envcp_padding	; move it to the next row
	set this_end			- + + [element] [element{2}] [envcp_padding] [main_sec]						; recalculate this_end, now that the element is on a new row
	set element				this_end{0}>main_sec{2} [. . 0 .]											; ... and cull if there's still not enough main_sec width to fit
	set this_y_end			+ + element{1} element{3} envcp_padding{1}									; find its bottom
	set element				this_y_end{0}>main_sec{3} [0] !previous_element{1} [0] .					; cull if there's not enough height to fit, or if previous_element was culled
	set main_sec_last 		element{3}>0 this_y_end{0} .

	; values readout control
	set valuesRoom			- - main_sec{3} this_y_end{0} envcp_padding{1}								; is the remaining height after this element enough for the values to fit?
	set hideValues			valuesRoom{0}<8 element{3}!=0 1 . 										; if not, and this element isn't already hidden, cull the values

	set previous_element element
endmacro
